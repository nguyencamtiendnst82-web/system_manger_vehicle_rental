-- =====================================================
-- DATABASE: QUAN LY CHO THUE XE
-- =====================================================

CREATE DATABASE IF NOT EXISTS quan_ly_cho_thue_xe
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE quan_ly_cho_thue_xe;


-- =====================================================
-- 1. BẢNG LOẠI XE
-- =====================================================

CREATE TABLE LOAI_XE (
    maLoaiXe INT AUTO_INCREMENT PRIMARY KEY,
    tenLoaiXe VARCHAR(100) NOT NULL,
    hangXe VARCHAR(100) NOT NULL
    
);


-- =====================================================
-- 2. BẢNG XE
-- =====================================================

CREATE TABLE XE (
    maXe INT AUTO_INCREMENT PRIMARY KEY,
    maLoaiXe INT NOT NULL,
    bienSoXe VARCHAR(20) NOT NULL UNIQUE,
    mauXe VARCHAR(50),
    soKmDaChay INT DEFAULT 0,
    trangThai ENUM(
    	'SAN_SANG',
    	'DANG_THUE',
    	'DANG_BAO_TRI',
    	'NGUNG_HOAT_DONG'
     ) DEFAULT 'SAN_SANG',
    soCho INT NOT NULL

    CONSTRAINT fk_xe_loaixe
        FOREIGN KEY (maLoaiXe)
        REFERENCES LOAI_XE(maLoaiXe)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 3. BẢNG NHÂN VIÊN
-- =====================================================

CREATE TABLE NHAN_VIEN (
    maNV INT AUTO_INCREMENT PRIMARY KEY,
    hoTen VARCHAR(100) NOT NULL,
    SDT VARCHAR(15),
    email VARCHAR(100)
);


-- =====================================================
-- 4. BẢNG KHÁCH HÀNG
-- =====================================================

CREATE TABLE KHACH_HANG (
    maKH INT AUTO_INCREMENT PRIMARY KEY,
    hoTen VARCHAR(100) NOT NULL,
    SDT VARCHAR(15),
    email VARCHAR(100),
    CCCD VARCHAR(20) UNIQUE,
    bangLai VARCHAR(20),
    diaChi VARCHAR(255)
);


-- =====================================================
-- 5. BẢNG HỢP ĐỒNG
-- =====================================================

CREATE TABLE HOP_DONG (
    maHD INT AUTO_INCREMENT PRIMARY KEY,

    maKH INT NOT NULL,
    maXe INT NOT NULL,
    maNV INT NOT NULL,

    ngayThue DATETIME NOT NULL,
    ngayTra DATETIME NOT NULL,
    ngayTraThucTe DATETIME,

    tienThue DECIMAL(15,2) DEFAULT 0,

    trangThai ENUM(
    	'CHO_XAC_NHAN',
    	'DA_XAC_NHAN',
    	'DANG_THUE',
    	'DA_TRA',
    	'DA_HUY'
     ) DEFAULT 'CHO_XAC_NHAN',

    CONSTRAINT fk_hd_khachhang
        FOREIGN KEY (maKH)
        REFERENCES KHACH_HANG(maKH)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_hd_xe
        FOREIGN KEY (maXe)
        REFERENCES XE(maXe)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_hd_nhanvien
        FOREIGN KEY (maNV)
        REFERENCES NHAN_VIEN(maNV)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 6. BẢNG THANH TOÁN
-- =====================================================

CREATE TABLE THANH_TOAN (
    maTT INT AUTO_INCREMENT PRIMARY KEY,

    maHD INT NOT NULL,
    soTien DECIMAL(15,2) NOT NULL,
    phuongThuc ENUM(
    	'TIEN_MAT',
    	'CHUYEN_KHOAN',
    	'THE'
    ) NOT NULL,
    ngayTT DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_tt_hopdong
        FOREIGN KEY (maHD)
        REFERENCES HOP_DONG(maHD)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 7. BẢNG BẢO TRÌ
-- =====================================================

CREATE TABLE BAO_TRI (
    maBT INT AUTO_INCREMENT PRIMARY KEY,

    maXe INT NOT NULL,
    noiDung TEXT,
    ngayBatDau DATE,
    ngayKetThuc DATE,
    chiPhi DECIMAL(15,2) DEFAULT 0,
    trangThai ENUM(
    	'CHO_BAO_TRI',
    	'DANG_BAO_TRI',
    	'DA_HOAN_THANH'
     ) DEFAULT 'CHO_BAO_TRI',

    CONSTRAINT fk_bt_xe
        FOREIGN KEY (maXe)
        REFERENCES XE(maXe)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 8. BẢNG TÌNH TRẠNG KHI TRẢ XE
-- =====================================================

CREATE TABLE TINH_TRANG_TRA_XE (
    maTinhTrang INT AUTO_INCREMENT PRIMARY KEY,

    maHD INT NOT NULL,

    ngayTra DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    tinhTrangXe TEXT,
    mucNhienLieu INT,
    soKmThucTe INT,
    ghiChu TEXT,

    CONSTRAINT fk_tttx_hopdong
        FOREIGN KEY (maHD)
        REFERENCES HOP_DONG(maHD)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);

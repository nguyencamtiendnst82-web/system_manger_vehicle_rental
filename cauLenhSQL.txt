-- =====================================================
-- DATABASE: QUAN LY CHO THUE XE
-- Yêu cầu: MySQL 8.0.16+ hoặc MariaDB 10.2.1+ (để CHECK có hiệu lực)
-- =====================================================

CREATE DATABASE IF NOT EXISTS manager_vehicle_rent
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE manager_vehicle_rent;


-- =====================================================
-- 1. BẢNG LOẠI XE
-- =====================================================

CREATE TABLE loai_xe (
    maLoaiXe INT AUTO_INCREMENT PRIMARY KEY,
    tenLoaiXe VARCHAR(100) NOT NULL,
    hangXe VARCHAR(100) NOT NULL,
    giaThueNgay DECIMAL(15,2) NOT NULL DEFAULT 0,

    CONSTRAINT chk_lx_gia CHECK (giaThueNgay >= 0)
);


-- =====================================================
-- 2. BẢNG XE
-- =====================================================

CREATE TABLE xe (
    maXe INT AUTO_INCREMENT PRIMARY KEY,
    maLoaiXe INT NOT NULL,
    bienSoXe VARCHAR(20) NOT NULL UNIQUE,
    mauXe VARCHAR(50),
    soKmDaChay INT DEFAULT 0,
    trangThai ENUM(
        'SAN_SANG',
        'DANG_THUE',
        'DANG_BAO_TRI',
        'NGUNG_HOAT_DONG'
    ) NOT NULL DEFAULT 'SAN_SANG',
    soCho INT NOT NULL,

    CONSTRAINT fk_xe_loaixe
        FOREIGN KEY (maLoaiXe)
        REFERENCES loai_xe(maLoaiXe)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 3. BẢNG NHÂN VIÊN
-- =====================================================

CREATE TABLE nhan_vien (
    maNV INT AUTO_INCREMENT PRIMARY KEY,
    hoTen VARCHAR(100) NOT NULL,
    SDT VARCHAR(15),
    email VARCHAR(100)
);


-- =====================================================
-- 4. BẢNG KHÁCH HÀNG
-- =====================================================

CREATE TABLE khach_hang (
    maKH INT AUTO_INCREMENT PRIMARY KEY,
    hoTen VARCHAR(100) NOT NULL,
    SDT VARCHAR(15),
    email VARCHAR(100),
    CCCD VARCHAR(20) UNIQUE,
    bangLai VARCHAR(20),
    diaChi VARCHAR(255)
);


-- =====================================================
-- 5. BẢNG HỢP ĐỒNG
-- =====================================================

CREATE TABLE hop_dong (
    maHD INT AUTO_INCREMENT PRIMARY KEY,

    maKH INT NOT NULL,
    maXe INT NOT NULL,
    maNV INT NOT NULL,

    ngayThue DATETIME NOT NULL,
    ngayTra DATETIME NOT NULL,
    ngayTraThucTe DATETIME,

    tienThue DECIMAL(15,2) NOT NULL DEFAULT 0,
    phiTraTre DECIMAL(15,2) NOT NULL DEFAULT 0,

    trangThai ENUM(
        'CHO_XAC_NHAN',
        'DA_XAC_NHAN',
        'DANG_THUE',
        'DA_TRA',
        'DA_HUY'
    ) NOT NULL DEFAULT 'CHO_XAC_NHAN',

    -- Index phục vụ kiểm tra trùng lịch; cũng là index cho khóa ngoại maXe
    INDEX hop_dong_maXe_ngayThue_ngayTra_idx (maXe, ngayThue, ngayTra),

    CONSTRAINT chk_hd_ngay CHECK (ngayTra > ngayThue),
    CONSTRAINT chk_hd_tien CHECK (tienThue >= 0),
    CONSTRAINT chk_hd_phitre CHECK (phiTraTre >= 0),

    CONSTRAINT fk_hd_khachhang
        FOREIGN KEY (maKH)
        REFERENCES khach_hang(maKH)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_hd_xe
        FOREIGN KEY (maXe)
        REFERENCES xe(maXe)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_hd_nhanvien
        FOREIGN KEY (maNV)
        REFERENCES nhan_vien(maNV)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 6. BẢNG THANH TOÁN
-- =====================================================

CREATE TABLE thanh_toan (
    maTT INT AUTO_INCREMENT PRIMARY KEY,

    maHD INT NOT NULL,
    soTien DECIMAL(15,2) NOT NULL,
    phuongThuc ENUM(
        'TIEN_MAT',
        'CHUYEN_KHOAN',
        'THE'
    ) NOT NULL,
    ngayTT DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_tt_sotien CHECK (soTien > 0),

    CONSTRAINT fk_tt_hopdong
        FOREIGN KEY (maHD)
        REFERENCES hop_dong(maHD)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 6b. BẢNG HOÀN TIỀN (hoàn lại khách khi hủy hợp đồng đã thanh toán)
-- =====================================================

CREATE TABLE hoan_tien (
    maHT INT AUTO_INCREMENT PRIMARY KEY,

    maHD INT NOT NULL,
    soTien DECIMAL(15,2) NOT NULL,
    lyDo VARCHAR(255),
    ngayHT DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    INDEX hoan_tien_maHD_idx (maHD),
    CONSTRAINT chk_ht_sotien CHECK (soTien > 0),

    CONSTRAINT fk_ht_hopdong
        FOREIGN KEY (maHD)
        REFERENCES hop_dong(maHD)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 7. BẢNG BẢO TRÌ
-- =====================================================

CREATE TABLE bao_tri (
    maBT INT AUTO_INCREMENT PRIMARY KEY,

    maXe INT NOT NULL,
    noiDung TEXT,
    ngayBatDau DATE,
    ngayKetThuc DATE,
    chiPhi DECIMAL(15,2) DEFAULT 0,
    trangThai ENUM(
        'CHO_BAO_TRI',
        'DANG_BAO_TRI',
        'DA_HOAN_THANH'
    ) NOT NULL DEFAULT 'CHO_BAO_TRI',

    CONSTRAINT chk_bt_chiphi CHECK (chiPhi IS NULL OR chiPhi >= 0),

    CONSTRAINT fk_bt_xe
        FOREIGN KEY (maXe)
        REFERENCES xe(maXe)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =====================================================
-- 8. BẢNG TÌNH TRẠNG KHI TRẢ XE (mỗi hợp đồng chỉ có 1 lần trả)
-- =====================================================

CREATE TABLE tinh_trang_tra_xe (
    maTinhTrang INT AUTO_INCREMENT PRIMARY KEY,

    maHD INT NOT NULL UNIQUE,

    ngayTra DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    tinhTrangXe TEXT,
    mucNhienLieu INT,
    soKmThucTe INT,
    ghiChu TEXT,

    CONSTRAINT chk_tttx_nhienlieu CHECK (mucNhienLieu IS NULL OR mucNhienLieu BETWEEN 0 AND 100),

    CONSTRAINT fk_tttx_hopdong
        FOREIGN KEY (maHD)
        REFERENCES hop_dong(maHD)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);
